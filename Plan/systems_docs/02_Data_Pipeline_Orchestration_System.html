4. Mã Nguồn Airflow DAG Python (`dags/ai_model_weekly_retrain.py`)

from airflow import DAG from airflow.operators.python import PythonOperator from airflow.operators.bash import BashOperator from datetime import datetime, timedelta import requests default_args = {'owner': 'data_eng', 'retries': 2, 'retry_delay': timedelta(minutes=5)} with DAG('ai_model_weekly_retrain', default_args=default_args, schedule_interval='0 2 * * 0', start_date=datetime(2026, 1, 1), catchup=False) as dag: t1 = BashOperator(task_id='check_db', bash_command='curl -f http://clickhouse_server:8123/ping || exit 1') def trigger_retrain(): res = requests.post("http://fastapi_serving:8000/api/v1/model/retrain", json={"model": "LSTM"}) if res.status_code != 200: raise ValueError("Retrain failed!") t2 = PythonOperator(task_id='trigger_retrain', python_callable=trigger_retrain) t1 >> t2
